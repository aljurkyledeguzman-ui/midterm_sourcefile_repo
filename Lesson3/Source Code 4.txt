// ================================================================
// PART A: STACK
// ================================================================

class Stack {
  // Initialize internal storage array
  constructor() {
    this.items = [];
  }

  // Push an element onto the top of the stack
  push(element) {
    this.items.push(element);
  }

  // Remove and return the top element
  pop() {
    if (this.isEmpty()) return null;
    return this.items.pop();
  }

  // Peek at the top element without removing it
  peek() {
    if (this.isEmpty()) return null;
    return this.items[this.items.length - 1];
  }

  // Check if the stack contains no elements
  isEmpty() {
    return this.items.length === 0;
  }

  // Return total count of elements
  size() {
    return this.items.length;
  }

  // Format array contents as a comma-separated string
  toString() {
    return this.items.join(', ');
  }
}

console.log('\nPART A: UNDO HISTORY');

const actionHistory = new Stack();

// 1. Push required actions
actionHistory.push('Opened Ticket #101');
actionHistory.push('Assigned Ticket #101');
actionHistory.push('Closed Ticket #101');

// 2. Display top action, undo action, remaining actions, and size
console.log(`Top Action: ${actionHistory.peek()}`);
console.log(`Undo Action: ${actionHistory.pop()}`);
console.log(`Remaining Actions: ${actionHistory.toString()}`);
console.log(`Stack Size: ${actionHistory.size()}`);

class Queue {
  // Initialize internal storage array
  constructor() {
    this.items = [];
  }

  // Add an element to the back of the queue
  enqueue(element) {
    this.items.push(element);
  }

  // Remove and return the front element
  dequeue() {
    if (this.isEmpty()) return null;
    return this.items.shift();
  }

  // View the front element without removing it
  front() {
    if (this.isEmpty()) return null;
    return this.items[0];
  }

  // Check if the queue contains no elements
  isEmpty() {
    return this.items.length === 0;
  }

  // Return total count of elements
  size() {
    return this.items.length;
  }

  // Format array contents as a comma-separated string
  toString() {
    return this.items.join(', ');
  }
}

// ================================================================

class PriorityQueue {
  // Initialize internal storage array
  constructor() {
    this.items = [];
  }

  // Insert item based on priority while preserving arrival order for equal priorities
  enqueue(request, priority) {
    const queueElement = { request, priority };
    let added = false;

    for (let i = 0; i < this.items.length; i++) {
      // Smaller number = higher priority (< keeps FIFO for equal priorities)
      if (queueElement.priority < this.items[i].priority) {
        this.items.splice(i, 0, queueElement);
        added = true;
        break;
      }
    }

    // If item has lowest/equal-lowest priority, push to end
    if (!added) {
      this.items.push(queueElement);
    }
  }

  // Process/remove highest priority (first) item
  dequeue() {
    if (this.size() === 0) return null;
    return this.items.shift();
  }

  // Return total item count
  size() {
    return this.items.length;
  }

  // Format each request and priority output
  print() {
    this.items.forEach((item) => {
      console.log(`${item.request}, Priority ${item.priority}`);
    });
  }
}

class Node {
  constructor(element) {
    this.element = element;
    this.next = null;
  }
}

class LinkedList {
  constructor() {
    this.length = 0;
    this.head = null;
  }

  // Append node to the end of list
  append(element) {
    const node = new Node(element);
    let current;

    if (this.head === null) {
      this.head = node;
    } else {
      current = this.head;
      while (current.next) {
        current = current.next;
      }
      current.next = node;
    }
    this.length++;
  }

  // Insert node at specific index/position
  insert(position, element) {
    if (position < 0 || position > this.length) return false;

    const node = new Node(element);
    let current = this.head;
    let previous;
    let index = 0;

    if (position === 0) {
      node.next = current;
      this.head = node;
    } else {
      while (index++ < position) {
        previous = current;
        current = current.next;
      }
      node.next = current;
      previous.next = node;
    }
    this.length++;
    return true;
  }

  // Remove node by position index
  removeAt(position) {
    if (position < 0 || position >= this.length) return null;

    let current = this.head;
    let previous;
    let index = 0;

    if (position === 0) {
      this.head = current.next;
    } else {
      while (index++ < position) {
        previous = current;
        current = current.next;
      }
      previous.next = current.next;
    }
    this.length--;
    return current.element;
  }

  // Return position index of given element
  indexOf(element) {
    let current = this.head;
    let index = 0;

    while (current) {
      if (current.element === element) return index;
      index++;
      current = current.next;
    }
    return -1;
  }

  // Remove node by value match using indexOf and removeAt
  remove(element) {
    const index = this.indexOf(element);
    return this.removeAt(index);
  }

  // Check if list has no nodes
  isEmpty() {
    return this.length === 0;
  }

  // Return node count
  size() {
    return this.length;
  }

  // Format node elements into a clean text list
  toString() {
    let current = this.head;
    let string = '';

    while (current) {
      string += current.element + (current.next ? '\n' : '');
      current = current.next;
    }
    return string;
  }
}
